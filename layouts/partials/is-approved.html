{{/* Partial: is-approved.html Outputs the string "true" (truthy) if the page should be shown on the main site. Outputs nothing/empty (falsy) if it should be hidden. Rules: - Non-state content (no 'states' or 'section' param): always shown. - State content: only shown if its slug appears in data/approved.yml. Usage: {{ if partial "is-approved.html" . }} ... {{ end }} */}} {{- $page := . -}} {{- $isState := or $page.Params.states $page.Params.section -}} {{- if not $isState -}} true {{- else -}} {{- $stateSlug := "" -}} {{- with $page.Params.states -}} {{- $stateSlug = index . 0 | urlize -}} {{- else -}} {{- $stateSlug = $page.Params.section | urlize -}} {{- end -}} {{- $section := $page.Section -}} {{- $slug := $page.File.Dir | path.Base -}} {{- $stateData := index site.Data.approved $stateSlug -}} {{- $sectionList := slice -}} {{- with $stateData -}} {{- $sectionList = index . $section | default slice -}} {{- end -}} {{- if in $sectionList $slug -}}true{{- end -}} {{- end -}}
